## 2. 方法调用

```                           javascript
let calculator = {
    operand1: 1,
    operand2: 1,
    add() {
        this.result = this.operand1 + this.operand2;
    }
};

calculator.add();       
calculator.result; // 2
```



**`this 关键字不具有变量那样的作用域机制，除了箭头函数，嵌套函数不会继承包含函数的 this 值。如果嵌套函数被当作方法来调用，那它的 this 值就是调用它的对象。如果嵌套函数（不是箭头函数）被当作函数来调用，则它的 this 值要么是全局对象（非严格模式），要么是 undefined（严格模式）。这里有一个常见的错误，就是对于定义在方法中的嵌套函数。如果将其当作函数来调用，以为可以使用 this 获得这个方法的调用上下文。以下代码演示了这个问题：`**

```javascript
let o = {
    m: function() {
        let self = this;
        this === o; // true
        f();
        
        function f() {
            this === o; // false，this 是全局对象或 undefined
            self === o; // true
        }
    }
};

o.m();
```



**`箭头函数可以继承 this 值：`**

```javascript
const f = () => {
    this === o; // true，因为箭头函数继承 this
};
```



**`调用嵌套函数的 bind() 方法，以定义一个在指定对象上被隐式调用的新函数：`**

```javascript
const f = (function() {
    this === o; // true
}).bind(this);
```

## 2. 剩余形参与可变长度实参列表

```javascript
function max(first=-Infinity, ...rest) {
    let maxValue = first;
    for(let n of rest) {
        if (n > maxValue) {
            maxValue = n;
        }
    }
    
    return maxValue;
}

max(1, 10, 100, 2, 3, 1000, 4, 5, 6); // 1000
```





## 3. Arguments 对象

```javascript
function max() {
    let maxValue = -Infinity;
    for(let i = 0; i < arguments.length; i++) {
        if (arguments[i] > maxValue) maxValue = arguments[i];
    }
    
    return maxValue;
}

max(1, 10, 100, 2, 3, 1000, 4, 5, 6); // 1000
```

## 6. 参数类型

```javascript
function sum(a) {
    let total = 0;
    for(let element of a) {
        if (typeof element !== "number") {
            throw new Error("sum()：elements must be numbers");
        }
        
        total += element;
    }
    
    return total;
}

sum([1, 2, 3]); // 6
sum(1, 2, 3); // TypeError：1 不是可迭代对象
sum([1, 2, "3"]); // TypeError： 元素 2 不是数值
```
