# exec() —— 在字符串中执行匹配，返回结果数组或 `null`

> 签名：`RegExp.prototype.exec(str): Array | null`
> 适合场景：需要匹配内容、捕获组、匹配位置时使用；`g` 标志下可配合循环逐个提取。

```javascript
// 1. 基本用法：匹配成功返回类数组
console.log(/dog/.exec("hotdog"));
// ["dog", index: 3, input: "hotdog", groups: undefined]

// 2. 无匹配时返回 null
console.log(/xyz/.exec("abcdef")); // null

// 3. 捕获组依次放在第 1、2、3... 位
const r3 = /(\d{4})-(\d{2})-(\d{2})/.exec("日期 2024-01-15");
console.log(r3[0]); // "2024-01-15"（整条匹配）
console.log(r3[1]); // "2024"
console.log(r3[2]); // "01"
console.log(r3[3]); // "15"

// 4. index 属性：匹配开始的位置
console.log(/cat/.exec("concatenate").index); // 3

// 5. input 属性：原始字符串
console.log(/cat/.exec("concatenate").input); // "concatenate"

// 6. 命名捕获组（ES2018）
const r6 = /(?<year>\d{4})-(?<month>\d{2})/.exec("2024-01");
console.log(r6.groups.year);  // "2024"
console.log(r6.groups.month); // "01"

// 7. g 标志下从 lastIndex 继续匹配，常配合 while 循环
const reg7 = /\d+/g;
const str7 = "a1b22c333";
let m7;
while ((m7 = reg7.exec(str7)) !== null) {
  console.log(`找到 ${m7[0]}，位置 ${m7.index}`);
}
// 找到 1，位置 1
// 找到 22，位置 3
// 找到 333，位置 6

// 8. 惰性（非贪婪）匹配：尽可能短地匹配
console.log(/<.+?>/.exec("<div><p>")[0]); // "<div>"

// 9. 多选分支 |
console.log(/cat|dog/.exec("hotdog")[0]); // "dog"

// 10. g 标志下连续调用 exec，lastIndex 前进
const reg10 = /a/g;
console.log(reg10.exec("aaa").index); // 0
console.log(reg10.exec("aaa").index); // 1（从上次的 lastIndex 继续）
```

## 注意事项

- `exec` 返回的数组额外带 `index`、`input`、`groups` 三个属性。
- `g` 标志下 `lastIndex` 是**状态化**的：匹配失败会重置为 0，混用 `test()` / `exec()` 容易踩坑。
- 现代开发中批量提取更推荐 `String.prototype.matchAll()`（见 04 篇）。

## 进阶例子（11–20）

```javascript
// 11. 反向引用匹配成对标签，提取标签名和内容
const m11 = /<(\w+)>([^<]*)<\/\1>/.exec("<div>box</div>");
console.log(m11[1], m11[2]); // "div" "box"

// 12. 命名捕获组解析 URL
const m12 = /^(?<proto>https?):\/\/(?<host>[^/]+)(?<path>\/.*)?$/.exec("https://a.com/x/y");
console.log(m12.groups); // { proto: "https", host: "a.com", path: "/x/y" }

// 13. 提取第一个 @ 用户名
console.log(/@(\w+)/.exec("hi @alice!")[1]); // "alice"

// 14. 提取货币金额（含可选小数）
const m14 = /([$€¥])(\d+(?:\.\d{1,2})?)/.exec("价格 $19.9 元");
console.log(m14[1], m14[2]); // "$" "19.9"

// 15. 非捕获组 (?:...) 不占组号
const m15 = /(?:\d{4})-(\d{2})/.exec("2024-01");
console.log(m15.length); // 2（整条匹配 + 1 个捕获组）
console.log(m15[1]);     // "01"

// 16. sticky（y 标志）：必须从 lastIndex 处开始匹配
const reg16 = /cat/y;
reg16.lastIndex = 3;
console.log(reg16.exec("concatenate")[0]); // "cat"（位置 3 正好是 cat）
reg16.lastIndex = 2;
console.log(reg16.exec("concatenate"));    // null（位置 2 不是 cat）

// 17. dotAll（s 标志）：让 . 也能匹配换行符
console.log(/a.b/s.exec("a\nb")); // ["a\nb", index: 0]
console.log(/a.b/.exec("a\nb"));  // null

// 18. 提取 rgb 颜色的三个数值
const m18 = /rgb\((\d+),\s*(\d+),\s*(\d+)\)/.exec("color: rgb(255, 128, 0)");
console.log(m18.slice(1).map(Number)); // [255, 128, 0]

// 19. 提取 Windows 路径的盘符、目录、文件名
const m19 = /^(\w):\\([^\\]+)\\([\w.]+)$/.exec("C:\\Users\\test.txt");
console.log(m19[1], m19[2], m19[3]); // "C" "Users" "test.txt"

// 20. 匹配失败时 lastIndex 自动重置为 0
const reg20 = /\d/g;
console.log(reg20.exec("abc")); // null（lastIndex 重置为 0）
console.log(reg20.exec("a1"));  // ["1"]（又能从头匹配）
```
