# replace() —— 替换匹配的部分

> 签名：`String.prototype.replace(regexp|substr, newSubstr|function): string`
> 适合场景：格式转换、脱敏、文本清洗。无 `g` 标志只替换**第一个**匹配。

```javascript
// 1. 无 g 标志：只替换第一个匹配
console.log("a-b-c".replace("-", "+")); // "a+b-c"

// 2. 有 g 标志：替换所有匹配
console.log("a-b-c".replace(/-/g, "+")); // "a+b+c"

// 3. 用 $1、$2 引用捕获组（日期格式转换）
console.log("2024-01-15".replace(/(\d{4})-(\d{2})-(\d{2})/, "$2/$3/$1")); // "01/15/2024"

// 4. $& 引用整条匹配
console.log("hello".replace(/\w+/, "[$&]")); // "[hello]"

// 5. 函数作为替换值：对匹配结果做计算
console.log("a1b2".replace(/\d/g, (d) => Number(d) * 2)); // "a2b4"

// 6. 手机号脱敏
console.log("13812345678".replace(/(\d{3})\d{4}(\d{4})/, "$1****$2")); // "138****5678"

// 7. 驼峰命名转短横线命名
console.log("backgroundColor".replace(/[A-Z]/g, (c) => "-" + c.toLowerCase())); // "background-color"

// 8. 短横线命名转驼峰命名
console.log("background-color".replace(/-([a-z])/g, (_, c) => c.toUpperCase())); // "backgroundColor"

// 9. 去除首尾空白
console.log("   hello   ".replace(/^\s+|\s+$/g, "")); // "hello"

// 10. 每个单词首字母大写
console.log("hello world foo".replace(/\b\w/g, (c) => c.toUpperCase())); // "Hello World Foo"
```

## 替换串中的特殊符号

| 符号 | 含义 |
| --- | --- |
| `$1`、`$2`... | 第 n 个捕获组 |
| `$&` | 整条匹配 |
| `` $` `` | 匹配之前的部分 |
| `$'` | 匹配之后的部分 |
| `$$` | 字面量 `$` |

## 进阶例子（11–20）

```javascript
// 11. 数字千分位格式化
console.log("1234567".replace(/\B(?=(\d{3})+(?!\d))/g, ",")); // "1,234,567"

// 12. 替换函数参与计算：摄氏转华氏
console.log("温度 25°C".replace(/(-?\d+)°C/g, (_, c) => `${(c * 9 / 5 + 32).toFixed(1)}°F`));
// "温度 77.0°F"

// 13. 简易模板引擎（数据里没有的键替换为空）
const data13 = { name: "张三", city: "北京" };
console.log("我是{{name}}，来自{{city}}".replace(/\{\{(\w+)\}\}/g, (_, k) => data13[k] ?? ""));
// "我是张三，来自北京"

// 14. Markdown 粗体转 <strong>（非贪婪 +）
console.log("hello **world** foo".replace(/\*\*(.+?)\*\*/g, "<strong>$1</strong>"));
// "hello <strong>world</strong> foo"

// 15. 去掉每行行首注释（m 标志让 ^ 匹配每一行开头）
console.log("// line1\n// line2".replace(/^\/\/\s*/gm, "")); // "line1\nline2"

// 16. $` 与 $'：匹配前、匹配后的内容
console.log("hello world".replace(/world/, "[$`]")); // "hello [hello ]"
console.log("hello world".replace(/hello/, "[$']")); // "[ world] world"

// 17. 邮箱脱敏：只保留用户名前两位
console.log("zhangsan@example.com".replace(/^(\w{2})\w+(@.*)$/, "$1***$2"));
// "zh***@example.com"

// 18. 交换两组内容（姓名倒置）
console.log("John Smith".replace(/(\w+)\s+(\w+)/, "$2, $1")); // "Smith, John"

// 19. 后行断言：保留首字母，其余打码
console.log("secret".replace(/(?<=\w)\w/g, "*")); // "s*****"

// 20. HTML 特殊字符转义（防 XSS 基础）
const esc20 = (s) => s.replace(/[<>&"]/g, (c) => ({ "<": "&lt;", ">": "&gt;", "&": "&amp;", '"': "&quot;" }[c]));
console.log(esc20('<script>alert("x")</script>'));
// "&lt;script&gt;alert(&quot;x&quot;)&lt;/script&gt;"
```
