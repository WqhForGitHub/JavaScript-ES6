# split() —— 按正则或字符串分割为数组

> 签名：`String.prototype.split(separator, limit?): string[]`
> 适合场景：解析 CSV、查询字符串、多分隔符切分。

```javascript
// 1. 按逗号分割
console.log("a,b,c".split(",")); // ["a", "b", "c"]

// 2. 按任意数量的空白分割
console.log("a  b   c".split(/\s+/)); // ["a", "b", "c"]

// 3. 第二个参数限制数组长度
console.log("a,b,c,d".split(",", 2)); // ["a", "b"]

// 4. 分隔符带捕获组时，分隔符也会出现在结果中
console.log("a,b,c".split(/(,)/)); // ["a", ",", "b", ",", "c"]

// 5. 多种分隔符一次搞定
console.log("a;b,c.d".split(/[;,.]/)); // ["a", "b", "c", "d"]

// 6. 按空字符串分割成字符数组
console.log("hello".split("")); // ["h", "e", "l", "l", "o"]

// 7. 按数字段分割（注意结尾可能产生空字符串）
console.log("a1b22c333".split(/\d+/)); // ["a", "b", "c", ""]

// 8. 兼容 \n 与 \r\n 两种换行
console.log("line1\nline2\r\nline3".split(/\r?\n/)); // ["line1", "line2", "line3"]

// 9. u 标志按 Unicode 码点分割（如 Emoji）
console.log("a🐱b".split(/[\u{1F300}-\u{1FAFF}]/u)); // ["a", "b"]

// 10. 空字符串分割返回 [""] 而不是 []
console.log("".split(",")); // [""]
```

## 注意事项

- 分隔符若出现在开头/结尾，结果首尾会出现空字符串 `""`（例 7），需要时用 `filter(Boolean)` 过滤。
- 正则里加**捕获组**可以保留分隔符（例 4），这是很多框架模板解析的基础技巧。

---

## 进阶例子（11–20）

```javascript
// 11. 解析简单 CSV 行
console.log("Tom,18,Beijing".split(",")); // ["Tom", "18", "Beijing"]

// 12. 分割路径（开头会产生空字符串，可用 filter 去掉）
console.log("/usr/local/bin".split("/"));                  // ["", "usr", "local", "bin"]
console.log("/usr/local/bin".split("/").filter(Boolean));  // ["usr", "local", "bin"]

// 13. 捕获组保留分隔符：把字母和数字拆开
console.log("a1b22c".split(/(\d+)/)); // ["a", "1", "b", "22", "c"]

// 14. 后行断言按句子分割（保留结尾标点）
console.log("Hi! How are you? I'm fine.".split(/(?<=[.!?])\s+/));
// ["Hi!", "How are you?", "I'm fine."]

// 15. 按空行分割段落
console.log("p1 line1\np1 line2\n\np2 line1".split(/\n{2,}/));
// ["p1 line1\np1 line2", "p2 line1"]

// 16. 前瞻断言：数字每 3 位一组
console.log("1234567".split(/\B(?=(?:\d{3})+(?!\d))/)); // ["1", "234", "567"]

// 17. 一次解析 a=1;b=2 这类配置串
console.log("a=1;b=2".split(/[;=]/)); // ["a", "1", "b", "2"]

// 18. 中英混排：按中文分割
console.log("hello你好world".split(/[\u4e00-\u9fa5]+/)); // ["hello", "world"]

// 19. limit 为 0 时返回空数组
console.log("a,b,c".split(",", 0)); // []

// 20. 按行分割并过滤空行（处理文件内容常用）
console.log("line1\n\nline2\n".split(/\r?\n/).filter(Boolean)); // ["line1", "line2"]
```
